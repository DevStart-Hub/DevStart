```{=html}
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Responsive PDF with Clickable Links</title>
  <!-- Load PDF.js from cdnjs -->
  <script src="https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.min.js"></script>
  <script src="https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.worker.min.js"></script>
  <style>
    /* Container for the PDF */
    #pdf-container {
      position: relative;
      width: 100%;
      border: 1px solid #ccc;
      overflow: hidden;
    }
    /* Canvas styling */
    canvas {
      display: block;
      width: 100%;
      height: auto;
    }
    /* Annotation layer styling */
    #annotation-layer {
      position: absolute;
      top: 0;
      left: 0;
      z-index: 10;
      /* The layer will be resized via JS to match the canvas */
    }
    /* Individual clickable link styling */
    .annotation-layer a {
      position: absolute;
      text-decoration: none;
      /* Uncomment below for debugging clickable areas */
      /* background: rgba(0, 0, 255, 0.2); */
      /* border: 1px solid rgba(0, 0, 255, 0.5); */
    }
  </style>
</head>
<body>
  <div id="pdf-container">
    <canvas id="pdf-canvas"></canvas>
    <!-- This div will hold the clickable link overlays -->
    <div id="annotation-layer" class="annotation-layer"></div>
  </div>

  <script>
    pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.worker.min.js';

    const url = '../../resources/GettingStartedWithDevStart/FlowchartPython.pdf';
    const container = document.getElementById('pdf-container');
    const canvas = document.getElementById('pdf-canvas');
    const annotationLayer = document.getElementById('annotation-layer');
    const context = canvas.getContext('2d');

    let pdfDoc, page, viewport;

    // Function to render the page with a computed scale that fits the container's width
    function renderPage() {
      // Clear previous annotation overlays
      annotationLayer.innerHTML = '';

      // Get container width
      const containerWidth = container.clientWidth;

      // First, get a viewport with scale=1 to know the intrinsic dimensions
      const initialViewport = page.getViewport({ scale: 1 });
      const scale = containerWidth / initialViewport.width;

      // Compute a new viewport with this scale
      viewport = page.getViewport({ scale: scale });
      canvas.width = viewport.width;
      canvas.height = viewport.height;

      // Resize the annotation layer to exactly overlay the canvas
      annotationLayer.style.width = canvas.width + 'px';
      annotationLayer.style.height = canvas.height + 'px';

      // Render the page onto the canvas
      const renderContext = {
        canvasContext: context,
        viewport: viewport
      };

      page.render(renderContext).promise.then(() => {
        // After rendering, get annotations (links)
        return page.getAnnotations();
      }).then(function(annotationsData) {
        annotationsData.forEach(function(annotation) {
          if (annotation.subtype === 'Link' && annotation.url) {
            // Normalize and convert annotation rectangle
            const rect = pdfjsLib.Util.normalizeRect(annotation.rect);
            const transform = viewport.convertToViewportRectangle(rect);
            const x = Math.min(transform[0], transform[2]);
            const y = Math.min(transform[1], transform[3]);
            const width = Math.abs(transform[0] - transform[2]);
            const height = Math.abs(transform[1] - transform[3]);

            const linkEl = document.createElement('a');
            linkEl.href = annotation.url;
            linkEl.target = '_blank';
            linkEl.style.left = x + 'px';
            linkEl.style.top = y + 'px';
            linkEl.style.width = width + 'px';
            linkEl.style.height = height + 'px';

            annotationLayer.appendChild(linkEl);
          }
        });
      });
    }

    // Load the PDF document and the first page
    pdfjsLib.getDocument(url).promise.then(function(pdf) {
      pdfDoc = pdf;
      return pdfDoc.getPage(1);
    }).then(function(pg) {
      page = pg;
      renderPage();
    }).catch(function(error) {
      console.error('Error loading PDF: ', error);
    });

    // Listen to window resize events and re-render the page accordingly
    window.addEventListener('resize', function() {
      if (page) {
        renderPage();
      }
    });
  </script>
</body>
</html>
```
